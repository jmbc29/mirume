# Mirume backend dependencies. Pinned versions live in backend/requirements.txt;
# this is the annotated overview. Install either one:
#   pip install -r requirements.txt            (from the repo root)
#   pip install -r ../requirements.txt         (from backend/)

# Web server
fastapi
uvicorn
watchfiles          # dev-only: event-based --reload watcher; without it uvicorn stat-polls the whole venv/ tree and can wedge the worker

# macOS Accessibility API + cursor / screenshot (accessibility.py, ocr.py)
pyobjc-framework-Cocoa
pyobjc-framework-ApplicationServices
pyobjc-framework-Quartz
pyobjc-framework-Vision   # screenshot OCR fallback for Chrome/sandboxed web content; built into macOS 13+, no model download

# Japanese NLP
fugashi             # MeCab tokeniser wrapper (tokeniser.py)
unidic-lite         # dictionary fugashi tokenises against
fasttext            # lid.176 language identification (language.py)

# Persistence
sqlalchemy
pydantic

# Translation (both optional at runtime — see backend/.env)
python-dotenv       # translator.py loads .env
deepl               # translator.py: DeepL English<->Japanese, when DEEPL_API_KEY is set
anthropic           # translator.py: Claude JA->EN sentence fallback, when ANTHROPIC_API_KEY is set

# Screenshot capture for the OCR fallback (ocr.py)
numpy               # <2; imported by fasttext
pillow

# Misc
requests            # dictionary / model / Tatoeba downloads
